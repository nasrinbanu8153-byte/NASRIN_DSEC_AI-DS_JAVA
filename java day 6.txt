1.MOVING ZEROS:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        int a[] = {0,5,0,3,8,0,2};
        int i = 0;
        int j = 0;
        for( i=0; i<a.length; i++){
            if(a[i] != 0){
                a[j] = a[i];
                j++;
            }
        }
        while(j < a.length){
            a[j] = 0;
            j++;
        }
        System.out.println("Array after moving zeros:");
        for(int l = 0; l < a.length; l++){
            System.out.println(a[l]+" ");
        }
    }
}
OUTPUT:
Array after moving zeros:
5 
3 
8 
2 
0 
0 
0 

2.UNIQUE NUMBERS:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        int a[] = {1,1,5,3,7,5};
        int n = a.length;
        

        System.out.println("Array of unique numbers:");
        for (int i = 0; i < n; i++) {
            boolean s = false;
            
            
            for (int j = 0; j < i; j++) {
                if (a[i] == a[j]) {
                    s = true;
                    break;
                }
            }
        if (!s){
            
            System.out.println(a[i]+" ");
    
        }
    }
}
}
OUTPUT:
Array of unique numbers:
1 
5 
3 
7

6.BINARY SEARCH:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a[] = {10,20,30,40,50};
        
        int key = 50;
        int low = 0;
        int high = a.length - 1;
        
        int pos = -1;
        
        while(low <= high){
            int mid = (low+high)/2;
            if(a[mid] == key){
                pos = mid;
                break;
            }else if(key > a[mid]){
                low = mid + 1;
            }else{
                high = mid - 1;
            }
        }
        if(pos != -1)
            System.out.println("found  at index"+pos);
        else
            System.out.println("not found");
        
        
        
        
       
}
}

OUTPUT:
found  at index4

7.LARGEST ELEMENT:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a[] =new int[5];
        System.out.println("enter 5 elements:");
        for(int i = 0; i<5; i++){
            a[i] = sc.nextInt();
        }
        int max = a[0];
        for(int i=1; i<5; i++){
            if(a[i]>max){
                max = a[i];
            }
        }
        System.out.println("Largest element " + max);
    }
}
OUTPUT:
enter 5 elements:
49 50 56 98 89 54
Largest element 98
