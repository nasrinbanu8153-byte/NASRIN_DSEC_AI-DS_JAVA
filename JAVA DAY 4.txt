1.SUM OF GIVEN DIGITS:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int sum = 0;
        while(n > 0){
            int digit =  n % 10;
            sum = sum + digit;
            n = n / 10;
            
        }
            
        System.out.println("sum of digits " + sum);
    }
}
OUTPUT:
6543
sum of digits 18


2.GCD:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int gcd = 1;
        int limit = Math.min(a , b);
        for(int i =1; i<=limit; i++){
            if(a % i == 0 && b % i == 0){
                gcd = i;
                
            }
        }
        System.out.println("GCD " + gcd);
    }
}
OUTPUT:
15
18
GCD 3
