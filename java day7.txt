1.BUBBLE SORT:
import java.util.*;
public class Main {
    public static void main(String[] args) {
        int a[] = {40,10,30,20};
        for(int i = 0; i < a.length-1; i++){
            for(int j = 0; j < a.length-1-i; j++){
                if(a[j]>a[j+1]){
                    int temp = a[j];
                    a[j]=a[j+1];
                    a[j+1] = temp;
                }
            }
        }
        for(int i = 0; i < a.length; i++)
            System.out.println(a[i]+" ");
   }
}
OUTPUT:
10 
20 
30 
40
